        const ChefHat = () => (

          <svg xmlns="http://www.w3.org/2000/svg" width="80" height="80" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round">

          <svg xmlns="http://www.w3.org/2000/svg" width="20" height="20" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round">

            <path d="M8 16H3v5"/>

        const ArrowLeft = () => (

          <svg xmlns="http://www.w3.org/2000/svg" width="16" height="16" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round">

            <line x1="19" y1="12" x2="5" y2="12"/>

            <polyline points="12 19 5 12 12 5"/>

          </svg>

        );

 

        function MealSuggestionApp() {

          const [currentMeal, setCurrentMeal] = useState(null);

          const [currentPhrase, setCurrentPhrase] = useState('');

          const [showDetails, setShowDetails] = useState(false);

 

          const generateSuggestion = () => {

            const randomMeal = meals[Math.floor(Math.random() * meals.length)];

            const randomPhrase = phrases[Math.floor(Math.random() * phrases.length)];

            setCurrentMeal(randomMeal);

            setCurrentPhrase(randomPhrase);

            setShowDetails(false);

          };

 

          const tryAgain = () => {

            generateSuggestion();

          };

 

          const showMoreDetails = () => {

            setShowDetails(true);

          };

 

          const goBack = () => {

            setShowDetails(false);

          };

 

          return (

            <div className="min-h-screen bg-gradient-to-br from-orange-50 to-red-50 p-4 flex items-center justify-center">

              <div className="max-w-md w-full">

                {!currentMeal ? (

                  <div className="text-center">

                    <div className="mb-8">

                      <div className="w-20 h-20 mx-auto text-red-600 mb-4">

                        <ChefHat />

                      </div>

                      <h1 className="text-4xl font-bold text-gray-800 mb-2">

                        What Should We Eat?

                      </h1>

                      <p className="text-gray-600">Press the button and let fate decide</p>

                    </div>

                    <button

                      onClick={generateSuggestion}

                      className="w-full bg-red-600 hover:bg-red-700 text-white font-bold py-6 px-8 rounded-2xl text-2xl shadow-lg transform transition hover:scale-105 active:scale-95"

                    >

                      What should we eat?

                    </button>

                  </div>

                ) : !showDetails ? (

                  <div className="bg-white rounded-2xl shadow-xl p-8">

                    <div className="text-center mb-6">

                      <p className="text-gray-600 text-lg mb-2">{currentPhrase}</p>

                      <h2 className="text-3xl font-bold text-gray-800 mb-6">

                        {currentMeal.name}

                      </h2>

                    </div>

                   

                    <div className="space-y-3">

                      <button

                        onClick={tryAgain}

                        className="w-full bg-red-600 hover:bg-red-700 text-white font-semibold py-4 px-6 rounded-xl flex items-center justify-center gap-2 shadow-md transform transition hover:scale-105 active:scale-95"

                      >

                        <RefreshCw />

                        No, try again

                      </button>

                     

                      <button

                        onClick={showMoreDetails}

                        className="w-full bg-green-600 hover:bg-green-700 text-white font-semibold py-4 px-6 rounded-xl shadow-md transform transition hover:scale-105 active:scale-95"

                      >

                        Tell me more about this

                      </button>

                    </div>

                  </div>

                ) : (

                  <div className="bg-white rounded-2xl shadow-xl p-8">

                    <button

                      onClick={goBack}

                      className="mb-4 text-gray-600 hover:text-gray-800 flex items-center gap-1 transition"

                    >

                      <ArrowLeft />

                      Back

                    </button>

                   

                    <h2 className="text-3xl font-bold text-gray-800 mb-6">

                      {currentMeal.name}

                    </h2>

                   

                    <div className="mb-6">

                      <h3 className="text-lg font-semibold text-gray-700 mb-3">Ingredients:</h3>

                      <ul className="space-y-1">

                        {currentMeal.ingredients.map((ingredient, index) => (

                          <li key={index} className="text-gray-600 flex items-start">

                            <span className="text-red-600 mr-2">•</span>

                            {ingredient}

                          </li>

                        ))}

                      </ul>

                    </div>

                   

                    <div className="mb-6">

                      <h3 className="text-lg font-semibold text-gray-700 mb-3">Preparation:</h3>

                      <p className="text-gray-600 leading-relaxed">{currentMeal.prep}</p>

                    </div>

                   

                    <button

                      onClick={tryAgain}

                      className="w-full bg-red-600 hover:bg-red-700 text-white font-semibold py-4 px-6 rounded-xl flex items-center justify-center gap-2 shadow-md transform transition hover:scale-105 active:scale-95"

                    >

                      <RefreshCw />

                      Actually, no. Try again

                    </button>

                  </div>

                )}

              </div>

            </div>

          );

        }

 

        ReactDOM.render(<MealSuggestionApp />, document.getElementById('root'));

    </script>
